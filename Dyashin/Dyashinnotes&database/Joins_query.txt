	dept_id SERIAL PRIMARY KEY,
	dept_name VARCHAR UNIQUE NOT NULL
);

CREATE TABLE workers.emp(
	emp_id SERIAL PRIMARY KEY,
	emp_name VARCHAR NOT NULL,
	dept_id INT,
	FOREIGN KEY(dept_id) REFERENCES workers.dept(dept_id)
);

--INNER JOIN - In both the table , matching value is printed
SELECT e.emp_id,e.emp_name,d.dept_name
FROM workers.emp e
INNER JOIN workers.dept d
ON e.dept_id=d.dept_id

--LEFT JOIN -- all the tables in the left is printed ,matching value in right printed
SELECT e.emp_id,e.emp_name,d.dept_name
FROM workers.emp e
LEFT JOIN workers.dept d
ON e.dept_id=d.dept_id

--RIGHT JOIN -- ALL the value in the right table printed,matching value in left printed
SELECT e.emp_id,e.emp_name,d.dept_name
FROM workers.emp e
RIGHT JOIN workers.dept d
ON e.dept_id=d.dept_id

--FULL JOIN - both the table value is printed
SELECT e.emp_id,e.emp_name,d.dept_name
FROM workers.emp e
FULL JOIN workers.dept d
ON e.dept_id=d.dept_id

--CROSS JOIN - all department name matches with all emp_name
SELECT e.emp_id,e.emp_name,d.dept_name
FROM workers.emp e
CROSS JOIN workers.dept d
