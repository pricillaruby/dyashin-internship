CREATE TABLE bank.account (
    acc_id SERIAL PRIMARY KEY,
    cus_id INT,
    balance NUMERIC CHECK (balance >= 0),
    acc_type TEXT CHECK (acc_type IN ('savings', 'fixed', 'current')) NOT NULL,
    created_date DATE DEFAULT CURRENT_DATE,
    FOREIGN KEY (cus_id) REFERENCES bank.customer(cus_id)
);

CREATE TABLE bank.transactions(
	trans_id SERIAL PRIMARY KEY,
	acc_id INT,
	credited_amount FLOAT CHECK(credited_amount>=0),
	trans_date DATE DEFAULT CURRENT_DATE,
	trans_type TEXT CHECK (trans_type IN ('deposit', 'withdrawal', 'transfer')) NOT NULL,
	FOREIGN KEY (acc_id) REFERENCES bank.account(acc_id)
);

1. List the names and account types of all customers

SELECT c.cus_name, a.acc_type
FROM bank.customer c
JOIN bank.account a
ON c.cus_id = a.cus_id

2.Find the accounts where balance > 50000

SELECT *FROM bank.account
WHERE balance>=50000

3.Count how many customers have current account

SELECT COUNT(*) 
FROM bank.account
Where acc_type='current'
