	customer_phoneno VARCHAR UNIQUE NOT NULL,

CREATE TABLE bank.accounts(
	account_id SERIAL PRIMARY KEY,
	customer_id INT,
	FOREIGN KEY(customer_id) REFERENCES bank.customer(customer_id),
	balance FLOAT CHECK(balance>=0),
	account_type VARCHAR NOT NULL CHECK(account_type IN('Savings','Fixed','Current')),
	created_date DATE DEFAULT CURRENT_DATE 
)

CREATE TABLE bank.transactions(
	transaction_id SERIAL PRIMARY KEY,
	account_id INT,
	FOREIGN KEY(account_id) REFERENCES bank.accounts(account_id),
	amount FLOAT CHECK(amount>0),
	transaction_type VARCHAR NOT NULL CHECK(transaction_type IN('Deposit','Withdraw','Transfer')),
	transaction_date DATE DEFAULT CURRENT_DATE
)
