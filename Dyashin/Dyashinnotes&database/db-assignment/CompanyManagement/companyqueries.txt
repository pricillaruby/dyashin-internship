-- 24.Display the department and project where maximum hours are worked. 

SELECT MAX(w.hours_worked),d.dept_name,p.project_name
FROM Company.Project p
JOIN Company.Department d
ON p.dept_id = d.dept_id
JOIN Company.WorksOn w
ON p.project_id = w.project_id
GROUP BY d.dept_name,p.project_name
ORDER BY MAX(w.hours_worked) DESC
LIMIT 1
